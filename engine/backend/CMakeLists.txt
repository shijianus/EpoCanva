cmake_minimum_required(VERSION 3.25)
get_filename_component(EPOCANVA_BACKEND_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Backend v1 uses the existing Windows runtime and Unicode adapter")
endif()
if(NOT TARGET epocanva_candidates)
  add_subdirectory("${EPOCANVA_BACKEND_ROOT}/engine/candidates" candidates-core)
endif()
if(NOT TARGET epocanva_fusion)
  add_subdirectory("${EPOCANVA_BACKEND_ROOT}/engine/fusion" fusion-core)
endif()
if(NOT TARGET epocanva_backend_session)
  add_subdirectory("${EPOCANVA_BACKEND_ROOT}/engine/backend_session" backend-session)
endif()
if(NOT TARGET epocanva_prediction_core)
  add_subdirectory("${EPOCANVA_BACKEND_ROOT}/engine/prediction" prediction-core)
endif()
if(NOT TARGET epocanva_lexical_service)
  add_subdirectory("${EPOCANVA_BACKEND_ROOT}/engine/lexical_service" lexical-service)
endif()
if(NOT TARGET epocanva_adaptive_ranking)
  add_subdirectory("${EPOCANVA_BACKEND_ROOT}/engine/ranking" adaptive-ranking)
endif()
if(NOT TARGET epocanva_engine_api)
  add_subdirectory("${EPOCANVA_BACKEND_ROOT}/engine/core" engine-core)
endif()
if(NOT TARGET epocanva_ja_composition_core)
  add_library(epocanva_ja_composition_core STATIC "${EPOCANVA_BACKEND_ROOT}/engine/languages/ja/romaji_composer.cc")
  target_compile_features(epocanva_ja_composition_core PUBLIC cxx_std_17)
endif()
if(NOT TARGET epocanva_ko_composition_core)
  add_library(epocanva_ko_composition_core STATIC "${EPOCANVA_BACKEND_ROOT}/engine/languages/ko/two_set_composer.cc")
  target_compile_features(epocanva_ko_composition_core PUBLIC cxx_std_17)
endif()
if(NOT TARGET epocanva_zh_pinyin_core)
  add_library(epocanva_zh_pinyin_core STATIC
    "${EPOCANVA_BACKEND_ROOT}/engine/languages/zh/pinyin_initialism.cc")
  target_include_directories(epocanva_zh_pinyin_core PUBLIC
    "${EPOCANVA_BACKEND_ROOT}" "${EPOCANVA_BACKEND_ROOT}/engine/languages/zh")
  target_link_libraries(epocanva_zh_pinyin_core PUBLIC epocanva_prediction_core)
  target_compile_features(epocanva_zh_pinyin_core PUBLIC cxx_std_17)
endif()
foreach(_composition_target epocanva_ja_composition_core epocanva_ko_composition_core
        epocanva_zh_pinyin_core)
  target_compile_options(${_composition_target} PRIVATE /utf-8 /permissive- /W4 /EHsc)
  set_target_properties(${_composition_target} PROPERTIES CXX_EXTENSIONS OFF)
endforeach()
if(NOT TARGET epocanva_windows_learning_repository)
  add_library(epocanva_windows_learning_repository STATIC
    "${EPOCANVA_BACKEND_ROOT}/engine/data_store/windows_learning_repository.cc")
  target_link_libraries(epocanva_windows_learning_repository PUBLIC epocanva_learning_core PRIVATE advapi32)
  target_compile_definitions(epocanva_windows_learning_repository PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_compile_features(epocanva_windows_learning_repository PUBLIC cxx_std_17)
endif()
add_library(epocanva_prediction_abbreviation STATIC
  "${EPOCANVA_BACKEND_ROOT}/engine/prediction/abbreviation.cc")
target_include_directories(epocanva_prediction_abbreviation PUBLIC "${EPOCANVA_BACKEND_ROOT}/engine/prediction")
target_compile_features(epocanva_prediction_abbreviation PUBLIC cxx_std_17)
target_compile_options(epocanva_prediction_abbreviation PRIVATE /utf-8 /permissive- /W4 /EHsc)
target_link_libraries(epocanva_prediction_abbreviation PUBLIC epocanva_prediction_core)
add_library(epocanva_backend STATIC backend_api.cc profile_api.inc context_text.h
  "${EPOCANVA_BACKEND_ROOT}/platforms/windows/product/product_io.cc")
target_include_directories(epocanva_backend PUBLIC "${EPOCANVA_BACKEND_ROOT}/engine/include"
  PRIVATE "${EPOCANVA_BACKEND_ROOT}" "${EPOCANVA_BACKEND_ROOT}/platforms/windows/product")
target_compile_features(epocanva_backend PUBLIC cxx_std_17)
target_compile_definitions(epocanva_backend PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_compile_options(epocanva_backend PRIVATE /utf-8 /permissive- /W4 /EHsc)
set_target_properties(epocanva_backend PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(epocanva_backend PUBLIC epocanva_engine_api epocanva_candidates PRIVATE epocanva_fusion epocanva_backend_session
  epocanva_prediction_core epocanva_prediction_abbreviation epocanva_lexical_service epocanva_windows_learning_repository
  epocanva_adaptive_ranking epocanva_ja_composition_core epocanva_ko_composition_core epocanva_zh_pinyin_core bcrypt)
add_subdirectory("${EPOCANVA_BACKEND_ROOT}/engine/japanese_decoder" japanese-decoder)
target_link_libraries(epocanva_backend PRIVATE epocanva_japanese_decoder)
option(EPOCANVA_BACKEND_WITH_FRENCH_DICTIONARY "Build the verified real Hunspell adapter" OFF)
if(EPOCANVA_BACKEND_WITH_FRENCH_DICTIONARY)
  add_subdirectory("${EPOCANVA_BACKEND_ROOT}/engine/french_dictionary" french-dictionary)
  target_link_libraries(epocanva_backend PRIVATE epocanva_french_dictionary)
  target_compile_definitions(epocanva_backend PRIVATE EPOCANVA_BACKEND_FRENCH=1)
endif()
