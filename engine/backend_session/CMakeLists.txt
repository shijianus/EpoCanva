cmake_minimum_required(VERSION 3.25)
if(NOT TARGET epocanva_learning_core)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../data_store" "${CMAKE_CURRENT_BINARY_DIR}/learning_core")
endif()
add_library(epocanva_backend_session STATIC session.cc)
target_include_directories(epocanva_backend_session PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}"
  "${CMAKE_CURRENT_LIST_DIR}/../candidates")
target_link_libraries(epocanva_backend_session PUBLIC epocanva_learning_core)
target_compile_features(epocanva_backend_session PUBLIC cxx_std_17)
set_target_properties(epocanva_backend_session PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(epocanva_backend_session PRIVATE /utf-8 /permissive- /W4 /EHsc)
endif()
