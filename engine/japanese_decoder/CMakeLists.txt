cmake_minimum_required(VERSION 3.25)
get_filename_component(_ja_repo "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_library(epocanva_japanese_decoder STATIC japanese_decoder.cc "${_ja_repo}/platforms/windows/product/product_io.cc")
target_include_directories(epocanva_japanese_decoder PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${_ja_repo}" "${_ja_repo}/platforms/windows/product")
target_link_libraries(epocanva_japanese_decoder PRIVATE epocanva_prediction_core bcrypt)
target_compile_definitions(epocanva_japanese_decoder PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_compile_features(epocanva_japanese_decoder PUBLIC cxx_std_17)
target_compile_options(epocanva_japanese_decoder PRIVATE /utf-8 /permissive- /W4 /EHsc)
set_target_properties(epocanva_japanese_decoder PROPERTIES CXX_EXTENSIONS OFF)
