# EpoCanva W1 -- product shell, product TSF DLL and their tests.
#
# NOT A STANDALONE PROJECT. Included by platforms/windows/engine_bridge when
# EPOCANVA_BUILD_PRODUCT=ON, so every target shares the /MT static CRT and the already
# built engine executor. The product DLL compiles the SAME TSF sources as the P3.1
# skeleton, with engine_host.cc and the product configuration, and no test seam.
if(NOT TARGET epocanva_engine_executor)
  message(FATAL_ERROR
    "platforms/windows/product must be added by platforms/windows/engine_bridge; it "
    "links epocanva_engine_executor.")
endif()

set(EPOCANVA_PRODUCT_VERSION "")
file(STRINGS "${CMAKE_CURRENT_LIST_DIR}/VERSION" EPOCANVA_PRODUCT_VERSION LIMIT_COUNT 1)
string(STRIP "${EPOCANVA_PRODUCT_VERSION}" EPOCANVA_PRODUCT_VERSION)
if(EPOCANVA_PRODUCT_VERSION STREQUAL "")
  message(FATAL_ERROR "platforms/windows/product/VERSION is empty")
endif()

set(EPOCANVA_PRODUCT_COMMIT "")
execute_process(
  COMMAND git -C "${CMAKE_CURRENT_LIST_DIR}" rev-parse --short HEAD
  OUTPUT_VARIABLE _product_commit
  OUTPUT_STRIP_TRAILING_WHITESPACE
  ERROR_QUIET
  RESULT_VARIABLE _product_git_status)
if(_product_git_status EQUAL 0)
  set(EPOCANVA_PRODUCT_COMMIT "${_product_commit}")
endif()

set(EPOCANVA_PRODUCT_LEXICON
    "${CMAKE_CURRENT_LIST_DIR}/../../../build/product-data/epocanva_en_alpha.tsv"
    CACHE FILEPATH
    "Product lexicon generated by tools/data/build_product_lexicon.py")
if(NOT EXISTS "${EPOCANVA_PRODUCT_LEXICON}")
  message(WARNING
    "EPOCANVA_PRODUCT_LEXICON does not exist: ${EPOCANVA_PRODUCT_LEXICON}\n"
    "Run tools/data/build_product_lexicon.py before the product engine test.")
endif()

set(_product_definitions
  _CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
set(_product_options /W4 /permissive- /EHsc /Zc:__cplusplus /utf-8)
set(_product_defines
  EPOCANVA_PRODUCT_VERSION="${EPOCANVA_PRODUCT_VERSION}"
  EPOCANVA_PRODUCT_COMMIT="${EPOCANVA_PRODUCT_COMMIT}")

set(_product_source_dir "${CMAKE_CURRENT_LIST_DIR}")
set(_tsf_source_dir "${CMAKE_CURRENT_LIST_DIR}/../tsf")
set(_bridge_source_dir "${CMAKE_CURRENT_LIST_DIR}/../engine_bridge")
set(_executor_source_dir "${CMAKE_CURRENT_LIST_DIR}/../engine_executor")
set(_input_source_dir "${CMAKE_CURRENT_LIST_DIR}/../input")
set(_text_source_dir "${CMAKE_CURRENT_LIST_DIR}/../text")

add_library(epocanva_product_state STATIC
  product_paths.cc
  product_paths.h
  product_io.cc
  product_io.h
  product_settings.cc
  product_settings.h)
target_compile_definitions(epocanva_product_state PRIVATE
  ${_product_definitions} ${_product_defines})
target_compile_options(epocanva_product_state PRIVATE ${_product_options})
target_include_directories(epocanva_product_state PUBLIC "${_product_source_dir}")
target_link_libraries(epocanva_product_state PUBLIC bcrypt shell32 ole32)

add_library(epocanva_product_core STATIC
  product_manifest.cc
  product_manifest.h
  product_diagnostics.cc
  product_diagnostics.h)
target_compile_definitions(epocanva_product_core PRIVATE ${_product_definitions})
target_compile_options(epocanva_product_core PRIVATE ${_product_options})
target_include_directories(epocanva_product_core
  PUBLIC "${_product_source_dir}"
  PRIVATE "${_tsf_source_dir}")
target_link_libraries(epocanva_product_core PUBLIC epocanva_product_state)

add_executable(EpoCanva WIN32
  app_main.cc
  settings_window.cc
  settings_window.h
  ui_ids.h
  "${_tsf_source_dir}/identifiers.cc"
  EpoCanva.rc)
target_compile_definitions(EpoCanva PRIVATE ${_product_definitions} ${_product_defines})
target_compile_options(EpoCanva PRIVATE ${_product_options})
target_include_directories(EpoCanva PRIVATE
  "${_product_source_dir}"
  "${_tsf_source_dir}")
target_link_libraries(EpoCanva PRIVATE
  epocanva_product_core comctl32 shell32 ole32 gdi32 user32 shcore)
# The manifest is merged by the linker: /manifest:embed already owns RT_MANIFEST id 1,
# so embedding a second manifest resource through the .rc would be a duplicate.
target_link_options(EpoCanva PRIVATE
  "/MANIFESTINPUT:${CMAKE_CURRENT_LIST_DIR}/EpoCanva.manifest")

add_library(epocanva_tsf_product SHARED
  "${_tsf_source_dir}/dll_main.cc"
  "${_tsf_source_dir}/identifiers.cc"
  "${_tsf_source_dir}/module_state.cc"
  "${_tsf_source_dir}/class_factory.cc"
  "${_tsf_source_dir}/text_service.cc"
  "${_tsf_source_dir}/engine_host.cc"
  "${_tsf_source_dir}/epocanva_tsf.def"
  product_engine_config.cc
  product_engine_config.h)
target_compile_definitions(epocanva_tsf_product PRIVATE
  ${_product_definitions}
  ${_product_defines}
  EPOCANVA_TSF_ENGINE_HOST=1
  EPOCANVA_TSF_PRODUCT_LAYOUT=1)
target_compile_options(epocanva_tsf_product PRIVATE ${_product_options})
target_include_directories(epocanva_tsf_product PRIVATE
  "${_product_source_dir}"
  "${_tsf_source_dir}"
  "${_bridge_source_dir}"
  "${_executor_source_dir}"
  "${_input_source_dir}"
  "${_text_source_dir}")
target_link_libraries(epocanva_tsf_product PRIVATE
  epocanva_engine_executor
  epocanva_product_state
  ole32 uuid bcrypt shell32)

# A predictable output directory so the staging script never has to parse a CMake
# target. Debug and Release are kept apart; only Release is packaged.
set(_product_bin_dir "${CMAKE_BINARY_DIR}/product/bin")
set(_product_bin_dir_debug "${CMAKE_BINARY_DIR}/product/bin-debug")
set_target_properties(EpoCanva epocanva_tsf_product PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${_product_bin_dir}"
  RUNTIME_OUTPUT_DIRECTORY_RELEASE "${_product_bin_dir}"
  RUNTIME_OUTPUT_DIRECTORY_DEBUG "${_product_bin_dir_debug}")
set_target_properties(epocanva_tsf_product PROPERTIES OUTPUT_NAME epocanva_tsf)

# W2: deployment snapshot, marker retirement and named-mutex coordination. The executable
# uses its own directory as the product root, so CMake copies the shipped data next to it.
set(_deployment_test_dir "${CMAKE_BINARY_DIR}/product/deployment-tests")
add_executable(epocanva_product_deployment_tests
  tests/deployment_tests.cc
  product_engine_config.cc
  "${_tsf_source_dir}/engine_host.cc"
  "${_tsf_source_dir}/module_state.cc")
target_compile_definitions(epocanva_product_deployment_tests PRIVATE
  ${_product_definitions} ${_product_defines})
target_compile_options(epocanva_product_deployment_tests PRIVATE ${_product_options})
target_include_directories(epocanva_product_deployment_tests PRIVATE
  "${_product_source_dir}"
  "${_tsf_source_dir}"
  "${_executor_source_dir}"
  "${_bridge_source_dir}"
  "${_input_source_dir}")
target_link_libraries(epocanva_product_deployment_tests PRIVATE
  epocanva_engine_executor
  epocanva_product_state
  bcrypt shell32 ole32)
set_target_properties(epocanva_product_deployment_tests PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${_deployment_test_dir}"
  RUNTIME_OUTPUT_DIRECTORY_RELEASE "${_deployment_test_dir}"
  RUNTIME_OUTPUT_DIRECTORY_DEBUG "${_deployment_test_dir}")
add_custom_command(TARGET epocanva_product_deployment_tests POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E make_directory
          "${_deployment_test_dir}/data/lexicon"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
          "${CMAKE_CURRENT_LIST_DIR}/../../../data/product/default.yaml"
          "${_deployment_test_dir}/data/default.yaml"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
          "${CMAKE_CURRENT_LIST_DIR}/../../../data/product/epocanva_english.schema.yaml"
          "${_deployment_test_dir}/data/epocanva_english.schema.yaml"
  COMMENT "Staging the product data next to the deployment tests")
if(EXISTS "${EPOCANVA_PRODUCT_LEXICON}")
  add_custom_command(TARGET epocanva_product_deployment_tests POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${EPOCANVA_PRODUCT_LEXICON}"
            "${_deployment_test_dir}/data/lexicon/epocanva_en_alpha.tsv")
else()
  message(WARNING
    "the deployment test will fail because the product lexicon is missing: "
    "${EPOCANVA_PRODUCT_LEXICON}")
endif()

enable_testing()
set(_product_assert
  "${CMAKE_CURRENT_LIST_DIR}/../../../tests/english_completion/assert_exit.cmake")
set(_product_run "${CMAKE_BINARY_DIR}/product/run")

add_test(
  NAME product.unit.settings_manifest_paths
  COMMAND ${CMAKE_COMMAND}
    -DEXE=$<TARGET_FILE:epocanva_product_unit_tests>
    -DCASE=product-unit
    -DEXPECTED_EXIT=0
    "-DMARKER=RESULT PASS"
    -DMARKER_POLICY=require
    "-DPASS_ARGS=--run-root;${_product_run}/unit"
    -DTIMEOUT_SECONDS=120
    -P ${_product_assert})
set_tests_properties(product.unit.settings_manifest_paths
  PROPERTIES TIMEOUT 180 LABELS "windows;product;settings;manifest;paths")

add_test(
  NAME product.integration.engine_settings
  COMMAND ${CMAKE_COMMAND}
    -DEXE=$<TARGET_FILE:epocanva_product_engine_tests>

    "-DMARKER=RESULT PASS"
