# Standalone or aggregate lexical provider verification, without Rime or ABI.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  cmake_minimum_required(VERSION 3.25)
  project(EpoCanvaLexicalServiceTests LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  enable_testing()
endif()
get_filename_component(EPOCANVA_LEXICAL_TEST_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  file(REAL_PATH "${EPOCANVA_LEXICAL_TEST_REPO}/build" _lexical_test_build_root)
  file(REAL_PATH "${CMAKE_BINARY_DIR}" _lexical_test_binary_root)
  if(WIN32)
    string(TOLOWER "${_lexical_test_build_root}" _lexical_test_build_root)
    string(TOLOWER "${_lexical_test_binary_root}" _lexical_test_binary_root)
  endif()
  string(FIND "${_lexical_test_binary_root}/" "${_lexical_test_build_root}/" _lexical_test_prefix)
  if(NOT _lexical_test_prefix EQUAL 0 OR _lexical_test_binary_root STREQUAL _lexical_test_build_root)
    message(FATAL_ERROR "Lexical test artifacts must stay under build/<preset>/.")
  endif()
endif()
if(NOT TARGET epocanva_lexical_service)
  add_subdirectory("${EPOCANVA_LEXICAL_TEST_REPO}/engine/lexical_service"
                   "${CMAKE_CURRENT_BINARY_DIR}/lexical-service")
endif()
add_executable(epocanva_lexical_service_tests main.cc)
target_link_libraries(epocanva_lexical_service_tests PRIVATE epocanva_lexical_service)
set_target_properties(epocanva_lexical_service_tests PROPERTIES
  CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(epocanva_lexical_service_tests PRIVATE /W4 /WX /permissive- /utf-8 /EHsc)
else()
  target_compile_options(epocanva_lexical_service_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

# Test artifacts are generated by CMake under the build tree. SHA-256 is CMake's
# independent implementation; provider/model runtime never writes these files.
set(_lexical_fixture_root "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
file(MAKE_DIRECTORY "${_lexical_fixture_root}")
set(_lexical_id_hello "5840500155624144132")
set(_lexical_payload "L\t${_lexical_id_hello}\ten\tword\thello\nU\t${_lexical_id_hello}\t10\n")
string(SHA256 _lexical_payload_hash "${_lexical_payload}")
string(REPEAT "a" 64 _lexical_source_hash)
set(_lexical_headers "EPOCANVA_PREDICTION_TSV\t1\nid_scheme\tfnv1a64-language-normalized-v1\nnormalization\tascii-lower-utf8-v1\nsource\ttest-fixture\nsource_sha256\t${_lexical_source_hash}\npayload_sha256\t${_lexical_payload_hash}\nlexemes\t1\nunigram_total\t10\nbigram_total\t0\ntrigram_total\t0\ndata\n")
file(GENERATE OUTPUT "${_lexical_fixture_root}/valid.tsv" CONTENT "${_lexical_headers}${_lexical_payload}" NEWLINE_STYLE UNIX)
file(GENERATE OUTPUT "${_lexical_fixture_root}/corrupt.tsv" CONTENT "${_lexical_headers}${_lexical_payload}invalid\n" NEWLINE_STYLE UNIX)
string(REPLACE "EPOCANVA_PREDICTION_TSV\t1" "EPOCANVA_PREDICTION_TSV\t2" _lexical_v2 "${_lexical_headers}")
file(GENERATE OUTPUT "${_lexical_fixture_root}/version.tsv" CONTENT "${_lexical_v2}${_lexical_payload}" NEWLINE_STYLE UNIX)
file(GENERATE OUTPUT "${_lexical_fixture_root}/binary.dat" CONTENT "EPOCANVA_PREDICTION_BINARY\t1\n" NEWLINE_STYLE UNIX)

set(_lexical_assert "${EPOCANVA_LEXICAL_TEST_REPO}/tests/infrastructure/assert_probe.cmake")
foreach(_case text aliases english french korean ranking budgets loaders concurrent)
  add_test(NAME lexical.service.${_case}
    COMMAND "${CMAKE_COMMAND}"
      "-DPROBE=$<TARGET_FILE:epocanva_lexical_service_tests>"
      "-DPROBE_ARG1=--${_case}" "-DPROBE_ARG2=${_lexical_fixture_root}"
      -DEXPECTED_EXIT=0 -DEXPECT_MARKER=TRUE "-DMARKER=RESULT PASS"
      -DTIMEOUT_SECONDS=120 "-DCASE=lexical-${_case}" -P "${_lexical_assert}")
  set_tests_properties(lexical.service.${_case} PROPERTIES
    TIMEOUT 150 WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}"
    LABELS "lexical;core;unit")
endforeach()
foreach(_negative forcefail bad-args)
  if(_negative STREQUAL "forcefail")
    set(_expected_exit 1)
  else()
    set(_expected_exit 2)
  endif()
  add_test(NAME lexical.service.${_negative}
    COMMAND "${CMAKE_COMMAND}"
      "-DPROBE=$<TARGET_FILE:epocanva_lexical_service_tests>"
      "-DPROBE_ARG1=--${_negative}" "-DPROBE_ARG2=${_lexical_fixture_root}"
      "-DEXPECTED_EXIT=${_expected_exit}" -DEXPECT_MARKER=FALSE
      "-DMARKER=RESULT PASS" "-DCASE=lexical-${_negative}" -P "${_lexical_assert}")
  set_tests_properties(lexical.service.${_negative} PROPERTIES
    TIMEOUT 30 WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}" LABELS "lexical;core;negative")
endforeach()
