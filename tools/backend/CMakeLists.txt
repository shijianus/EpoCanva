cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(EpoCanvaBackend LANGUAGES C CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
  enable_testing()
endif()
get_filename_component(EPOCANVA_BACKEND_REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT TARGET epocanva_backend)
  add_subdirectory("${EPOCANVA_BACKEND_REPO}/engine/backend" backend)
endif()
add_executable(epocanva_backend_cli main.cc)
target_link_libraries(epocanva_backend_cli PRIVATE epocanva_backend epocanva_prediction_core)
target_include_directories(epocanva_backend_cli PRIVATE "${EPOCANVA_BACKEND_REPO}")
target_compile_features(epocanva_backend_cli PRIVATE cxx_std_17)
target_compile_options(epocanva_backend_cli PRIVATE /utf-8 /permissive- /W4 /EHsc)
add_subdirectory("${EPOCANVA_BACKEND_REPO}/tests/backend" backend-tests)
add_subdirectory("${EPOCANVA_BACKEND_REPO}/tests/backend_session" backend-session-tests)
add_subdirectory("${EPOCANVA_BACKEND_REPO}/tests/prediction_core" prediction-tests)
add_subdirectory("${EPOCANVA_BACKEND_REPO}/tests/prediction_abbreviation" abbreviation-tests)
add_subdirectory("${EPOCANVA_BACKEND_REPO}/tests/lexical_service" lexical-tests)
add_subdirectory("${EPOCANVA_BACKEND_REPO}/tests/japanese_decoder" japanese-tests)
if(EPOCANVA_BACKEND_WITH_FRENCH_DICTIONARY)
  add_subdirectory("${EPOCANVA_BACKEND_REPO}/tests/french_dictionary" french-dictionary-tests)
endif()
